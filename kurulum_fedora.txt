bash -c "$(cat <<'KAVRAM_KURULUM'
# KAVRAM KURULUM — Fedora / Nobara / RHEL türevleri (AlmaLinux, Rocky, CentOS Stream)
# Kullanım: kavram klasöründe terminal açın, bu dosyanın TÜM içeriğini kopyalayıp yapıştırın, Enter'a basın.
#           (veya:  bash kurulum_fedora.txt )
# Yapılanlar: sistem paketleri -> sanal ortam (venv) -> Python kütüphaneleri -> C++ derlemeleri
#             -> PAKETLEME (dist/ klasöründe kavram/kavram çalıştırılabilir dosyası + .tar.gz arşivi)
#             -> Kavram otomatik başlar.
# Not: motorlar her işlemcide çalışsın diye taşınabilir (x86-64) derlenir; yalnızca bu bilgisayarda en yüksek
#      hız için önce  export KAVRAM_MARCH=native  yazın.
[ -f kavram.py ] || { echo "HATA: Bu komutu kavram klasöründe (kavram.py'nin yanında) çalıştırın."; exit 1; }
export KAVRAM_DIR="$PWD"
export KAVRAM_OS=linux
export KAVRAM_MARCH="${KAVRAM_MARCH:-x86-64}"
liste() { cat */derleme.txt 2>/dev/null | sed -n "s/^# $1: *//p" | tr ' ' '\n' | sed '/^$/d' | sort -u | tr '\n' ' '; }

echo "==> Sistem paketleri kuruluyor (dnf, sudo parolası istenebilir)"
. /etc/os-release
# FFmpeg / GStreamer tam kodek desteği için RPM Fusion gerekir (Fedora'nın kendi deposunda sınırlı kodek vardır).
if [ "$ID" = "fedora" ] || [[ " $ID_LIKE " == *" fedora "* && "$ID" != "rhel" && "$ID" != "almalinux" && "$ID" != "rocky" && "$ID" != "centos" ]]; then
  FV=$(rpm -E %fedora)
  sudo dnf install -y "https://mirrors.rpmfusion.org/free/fedora/rpmfusion-free-release-$FV.noarch.rpm" \
                      "https://mirrors.rpmfusion.org/nonfree/fedora/rpmfusion-nonfree-release-$FV.noarch.rpm" \
    || echo "Uyarı: RPM Fusion eklenemedi; sınırlı kodekli ffmpeg-free kullanılacak."
else
  RV=$(rpm -E %rhel)
  sudo dnf install -y epel-release || true
  sudo dnf config-manager --set-enabled crb 2>/dev/null || sudo dnf config-manager --set-enabled powertools 2>/dev/null || true
  sudo dnf install -y "https://mirrors.rpmfusion.org/free/el/rpmfusion-free-release-$RV.noarch.rpm" \
                      "https://mirrors.rpmfusion.org/nonfree/el/rpmfusion-nonfree-release-$RV.noarch.rpm" \
    || echo "Uyarı: RPM Fusion eklenemedi; ffmpeg bulunamayabilir."
fi
if rpm -q ffmpeg-free >/dev/null 2>&1 || ! rpm -q ffmpeg >/dev/null 2>&1; then
  sudo dnf swap -y ffmpeg-free ffmpeg --allowerasing 2>/dev/null || sudo dnf install -y ffmpeg --allowerasing || true
fi
for p in $(liste dnf) $(liste dnf-opsiyonel); do
  sudo dnf install -y "$p" || echo "Uyarı: paket kurulamadı: $p"
done
rpm -q ffmpeg-devel >/dev/null 2>&1 || sudo dnf install -y ffmpeg-free-devel --allowerasing || echo "Uyarı: ffmpeg geliştirme paketi kurulamadı (converter motoru derlenemeyebilir)."

echo
echo "==> Sanal ortam (venv) hazırlanıyor"
python3 -m venv venv || { echo "HATA: venv oluşturulamadı"; exit 1; }
./venv/bin/pip install --upgrade pip wheel || { echo "HATA: pip güncellenemedi"; exit 1; }
PIP=$(liste pip)
./venv/bin/pip install $PIP pyinstaller || { echo "HATA: Python kütüphaneleri kurulamadı"; exit 1; }
./venv/bin/python -c "import sys; sys.exit(0 if sys.version_info >= (3, 13) else 1)" && ./venv/bin/pip install audioop-lts || true

echo
echo "==> Linux'a özel program dosyaları denetleniyor (rec.py, text.py ... varsayılan Linux dosyalarıdır)"
./venv/bin/python yapilandir.py || echo "Uyarı: yapilandir.py çalışmadı (var olan dosyalar kullanılacak)."

echo
echo "==> C++ motorları derleniyor (her program kendi klasöründe)"
HATALI=""
for d in */; do
  d=${d%/}
  [ -f "$d/derleme.txt" ] || continue
  echo "--- $d"
  ( cd "$d" && bash derleme.txt ) || HATALI="$HATALI $d"
done
mkdir -p export veri
if [ -n "$HATALI" ]; then
  echo "UYARI: Şu programların C++ derlemesi başarısız oldu:$HATALI"
  echo "       Hata ayrıntısı yukarıda. Diğer programlar çalışmaya devam eder."
fi

echo
echo "==> Paketleniyor (PyInstaller) -> dist/"
if bash paketle.sh --sadece-paket; then
  echo "    Paket hazır: dist/kavram/kavram   (arşiv: dist/*.tar.gz)"
  echo "    Kullanıcı olarak kurmak için:  bash paketle.sh --sadece-paket --kur"
else
  echo "UYARI: Paketleme başarısız oldu (yukarıdaki hataya bakın). Kavram yine de kaynak koddan başlatılacak."
fi

echo "==> Kurulum bitti, Kavram başlatılıyor..."
echo "    Sonraki açılışlar için kavram klasöründe:  ./venv/bin/python kavram.py   veya   dist/kavram/kavram"
exec ./venv/bin/python kavram.py
KAVRAM_KURULUM
)"
